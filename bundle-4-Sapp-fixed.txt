package com.sapp.planner;

import android.Manifest;
import android.app.Activity;
import android.app.AlarmManager;
import android.app.NotificationManager;
import android.content.ContentValues;
import android.content.Intent;
import android.content.SharedPreferences;
import android.content.pm.PackageManager;
import android.graphics.Color;
import android.graphics.Insets;

import android.os.Build;
import android.os.Bundle;

import android.provider.MediaStore;

import android.view.View;

import android.view.WindowInsetsController;
import android.webkit.JavascriptInterface;
import android.webkit.ValueCallback;
import android.webkit.WebChromeClient;
import android.webkit.WebResourceRequest;
import android.webkit.WebResourceResponse;
import android.webkit.WebSettings;
import android.webkit.WebView;
import android.webkit.WebViewClient;
import android.widget.Toast;

import androidx.core.content.FileProvider;
import androidx.webkit.WebViewAssetLoader;

import java.io.File;

public class MainActivity extends Activity {
    static final String HOST = "appassets.androidplatform.net";
    static volatile boolean fg = false;

    private WebView web;
    private ValueCallback<Uri[]> picker;
    private Uri cameraImageUri;

    @Override
    protected void onCreate(Bundle b) {
        super.onCreate(b);
        ReminderReceiver.channels(this);
        if (Build.VERSION.SDK_INT >= 33
            && checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED) {
            requestPermissions(new String[]{Manifest.permission.POST_NOTIFICATIONS}, 1);
        }
        if (checkSelfPermission(Manifest.permission.CAMERA) != PackageManager.PERMISSION_GRANTED) {
            requestPermissions(new String[]{Manifest.permission.CAMERA}, 2);
        }

        web = new WebView(this);
        web.setBackgroundColor(Color.parseColor("#f4f4f6"));
        setContentView(web);

        web.setOnApplyWindowInsetsListener((v, ins) -> {
            if (Build.VERSION.SDK_INT >= 30) {
                Insets sysBars = ins.getInsets(WindowInsets.Type.systemBars());
                Insets ime = ins.getInsets(WindowInsets.Type.ime());
                int bottomPadding = Math.max(sysBars.bottom, ime.bottom);
                v.setPadding(sysBars.left, sysBars.top, sysBars.right, bottomPadding);
            } else {
                v.setPadding(ins.getSystemWindowInsetLeft(), ins.getSystemWindowInsetTop(),
                    ins.getSystemWindowInsetRight(), ins.getSystemWindowInsetBottom());
            }
            return ins;
        });

        WebSettings s = web.getSettings();
        s.setJavaScriptEnabled(true);
        s.setDomStorageEnabled(true);
        s.setAllowFileAccess(true);
        s.setAllowContentAccess(true);
        s.setSupportZoom(false);
        s.setBuiltInZoomControls(false);
        s.setDisplayZoomControls(false);
        s.setMediaPlaybackRequiresUserGesture(false);
        web.setOverScrollMode(WebView.OVER_SCROLL_NEVER);
        web.addJavascriptInterface(new Bridge(), "Android");

        final WebViewAssetLoader loader = new WebViewAssetLoader.Builder()
            .addPathHandler("/assets/", new WebViewAssetLoader.AssetsPathHandler(this)).build();
        web.setWebViewClient(new WebViewClient() {
            @Override
            public WebResourceResponse shouldInterceptRequest(WebView v, WebResourceRequest r) {
                return loader.shouldInterceptRequest(r.getUrl());
            }
            @Override
            public boolean shouldOverrideUrlLoading(WebView v, WebResourceRequest r) {
                return !HOST.equals(r.getUrl().getHost());
            }
        });

        web.setWebChromeClient(new WebChromeClient() {
            @Override
            public boolean onShowFileChooser(WebView v, ValueCallback<Uri[]> cb, FileChooserParams p) {
                if (picker != null) picker.onReceiveValue(null);
                picker = cb;

                boolean isCapture = p.isCaptureEnabled();
                try {
                    if (isCapture) {
                        File cachePath = new File(getCacheDir(), "images");
                        cachePath.mkdirs();
                        File photoFile = new File(cachePath, "scan_" + System.currentTimeMillis() + ".jpg");
                        cameraImageUri = FileProvider.getUriForFile(MainActivity.this, "com.sapp.planner.fileprovider", photoFile);
                        Intent takePictureIntent = new Intent(MediaStore.ACTION_IMAGE_CAPTURE);
                        takePictureIntent.putExtra(MediaStore.EXTRA_OUTPUT, cameraImageUri);
                        startActivityForResult(takePictureIntent, 100);
                        return true;
                    }
                    Intent intent = p.createIntent();
                    startActivityForResult(intent, 101);
                    return true;
                } catch (Exception e) {
                    if (picker != null) {
                        picker.onReceiveValue(null);
                        picker = null;
                    }
                    return false;
                }
            }
        });

        web.loadUrl("https://" + HOST + "/assets/index.html");
    }

    @Override
    protected void onActivityResult(int rq, int rs, Intent d) {
        if (picker == null) return;
        if (rq == 100) {
            if (rs == RESULT_OK && cameraImageUri != null) {
                picker.onReceiveValue(new Uri[]{cameraImageUri});
            } else {
                picker.onReceiveValue(null);
            }
            picker = null;
        } else if (rq == 101) {
            picker.onReceiveValue(WebChromeClient.FileChooserParams.parseResult(rs, d));
            picker = null;
        }
    }

    @Override
    protected void onResume() {
        super.onResume();
        fg = true;
    }

    @Override
    protected void onPause() {
        fg = false;
        if (web != null) web.evaluateJavascript("window.__flush&&__flush()", null);
        super.onPause();
    }

    private void toast(final String m) {
        runOnUiThread(() -> Toast.makeText(this, m, Toast.LENGTH_LONG).show());
    }

    class Bridge {
        @JavascriptInterface public boolean isPro() { return true; }
        @JavascriptInterface public String price() { return "Free"; }
        @JavascriptInterface public void subscribe() {}
        @JavascriptInterface public void refresh() {}
        @JavascriptInterface public void sync(String json) { ReminderReceiver.schedule(MainActivity.this, json); }
        @JavascriptInterface public void timerStart(double end, String label) {
            ReminderReceiver.timer(MainActivity.this, (long) end, label);
        }
        @JavascriptInterface public void timerCancel() { ReminderReceiver.cancelTimer(MainActivity.this); }
        @JavascriptInterface public void welcome(String t, String b) {
            ReminderReceiver.show(MainActivity.this, 7001, t, b, false);
        }
        @JavascriptInterface public String tones() { return ReminderReceiver.tones(MainActivity.this); }
        @JavascriptInterface public void setTones(String timerUri, String notifUri) { 
            ReminderReceiver.setTones(MainActivity.this, timerUri, notifUri); 
        }
        @JavascriptInterface public String statusTones() { return ReminderReceiver.statusTones(MainActivity.this); }

        @JavascriptInterface public String status() {
            NotificationManager nm = (NotificationManager) getSystemService(NOTIFICATION_SERVICE);
            AlarmManager am = (AlarmManager) getSystemService(ALARM_SERVICE);
            PowerManager pm = (PowerManager) getSystemService(POWER_SERVICE);
            boolean n = nm.areNotificationsEnabled();
            boolean ex = Build.VERSION.SDK_INT < 31 || am.canScheduleExactAlarms();
            boolean bat = pm.isIgnoringBatteryOptimizations(getPackageName());
            return "{\"v\":4,\"notif\":" + n + ",\"exact\":" + ex + ",\"battery\":" + bat + "}";
        }

        @JavascriptInterface public void open(final String what) {
            runOnUiThread(() -> {
                try {
                    Intent i;
                    String pkg = getPackageName();
                    if ("notif".equals(what)) {
                        i = new Intent(Settings.ACTION_APP_NOTIFICATION_SETTINGS).putExtra(Settings.EXTRA_APP_PACKAGE, pkg);
                    } else if ("battery".equals(what)) {
                        i = new Intent(Settings.ACTION_REQUEST_IGNORE_BATTERY_OPTIMIZATIONS, Uri.parse("package:" + pkg));
                    } else if ("exact".equals(what) && Build.VERSION.SDK_INT >= 31) {
                        i = new Intent(Settings.ACTION_REQUEST_SCHEDULE_EXACT_ALARM, Uri.parse("package:" + pkg));
                    } else {
                        i = new Intent(Settings.ACTION_APPLICATION_DETAILS_SETTINGS, Uri.parse("package:" + pkg));
                    }
                    i.addFlags(Intent.FLAG_ACTIVITY_NEW_TASK);
                    startActivity(i);
                } catch (Exception ignored) { }
            });
        }

        @JavascriptInterface public void setBars(final String hex, final boolean dark) {
            runOnUiThread(() -> {
                try {
                    int c = Color.parseColor(hex);
                    getWindow().getDecorView().setBackgroundColor(c);
                    web.setBackgroundColor(c);
                    if (Build.VERSION.SDK_INT >= 30) {
                        WindowInsetsController wc = getWindow().getInsetsController();
                        if (wc != null) {
                            int m = WindowInsetsController.APPEARANCE_LIGHT_STATUS_BARS
                                | WindowInsetsController.APPEARANCE_LIGHT_NAVIGATION_BARS;
                            wc.setSystemBarsAppearance(dark ? 0 : m, m);
                        }
                    }
                } catch (Exception ignored) { }
            });
        }

        @JavascriptInterface public void saveIcs(String name, String text) {
            try {
                if (Build.VERSION.SDK_INT < 29) { toast("Requires Android 10+"); return; }
                ContentValues v = new ContentValues();
                v.put(MediaStore.Downloads.DISPLAY_NAME, name);
                v.put(MediaStore.Downloads.MIME_TYPE, "text/calendar");
                v.put(MediaStore.Downloads.RELATIVE_PATH, Environment.DIRECTORY_DOWNLOADS);
                Uri u = getContentResolver().insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI, v);
                OutputStream o = getContentResolver().openOutputStream(u);
                o.write(text.getBytes("UTF-8"));
                o.close();
                Intent i = new Intent(Intent.ACTION_VIEW);
                i.setDataAndType(u, "text/calendar");
                i.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION | Intent.FLAG_ACTIVITY_NEW_TASK);
                startActivity(i);
            } catch (Exception e) {
                toast("Saved to Downloads.");
            }
        }
    }

package com.sapp.planner;

import android.app.AlarmManager;
import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.content.SharedPreferences;
import android.database.Cursor;
import android.media.AudioAttributes;
import android.media.RingtoneManager;
import android.net.Uri;
import android.os.Build;

import org.json.JSONArray;
import org.json.JSONObject;

public class ReminderReceiver extends BroadcastReceiver {
    static final String CH_ALARM = "alarm_sapp_1";
    static final String CH_INFO = "info_sapp_1";
    static final int TIMER_ID = 9001;

    static Uri toneUri(Context c, String key, int type) {
        String s = c.getSharedPreferences("r", Context.MODE_PRIVATE).getString(key, "");
        if (s != null && !s.isEmpty()) {
            try { return Uri.parse(s); } catch (Exception ignored) { }
        }
        return RingtoneManager.getDefaultUri(type);
    }

    static void channels(Context c) {
        if (Build.VERSION.SDK_INT < 26) return;
        NotificationManager nm = (NotificationManager) c.getSystemService(Context.NOTIFICATION_SERVICE);
        AudioAttributes aa = new AudioAttributes.Builder()
            .setUsage(AudioAttributes.USAGE_ALARM)
            .setContentType(AudioAttributes.CONTENT_TYPE_SONIFICATION).build();
        NotificationChannel a = new NotificationChannel(CH_ALARM, "Timer and reminders", NotificationManager.IMPORTANCE_HIGH);
        a.setSound(toneUri(c, "timerTone", RingtoneManager.TYPE_ALARM), aa);
        a.enableVibration(true);
        a.setVibrationPattern(new long[]{0, 500, 250, 500, 250, 500});
        nm.createNotificationChannel(a);
        NotificationChannel i = new NotificationChannel(CH_INFO, "Notifications", NotificationManager.IMPORTANCE_DEFAULT);
        i.setSound(toneUri(c, "notifTone", RingtoneManager.TYPE_NOTIFICATION), aa);
        nm.createNotificationChannel(i);
    }

    static String tones(Context c) {
        JSONArray out = new JSONArray();
        Cursor cur = null;
        try {
            cur = RingtoneManager.getCursor(c, RingtoneManager.TYPE_NOTIFICATION);
            while (cur.moveToNext()) {
                String title = cur.getString(RingtoneManager.TITLE_COLUMN_INDEX);
                String uri = cur.getString(RingtoneManager.URI_COLUMN_INDEX);
                String id = cur.getString(RingtoneManager.ID_COLUMN_INDEX);
                JSONObject o = new JSONObject();
                o.put("title", title == null ? "Notification Sound" : title);
                o.put("uri", uri + "/" + id);
                out.put(o);
            }
        } catch (Exception ignored) {
        } finally { if (cur != null) cur.close(); }
        return out.toString();
    }

    static String statusTones(Context c) {
        SharedPreferences sp = c.getSharedPreferences("r", Context.MODE_PRIVATE);
        return "{\"timer\":"+JSONObject.quote(sp.getString("timerTone", ""))+ ",\"notif\":"+JSONObject.quote(sp.getString("notifTone", ""))+"}";
    }

    static void setTones(Context c, String timerUri, String notifUri) {
        if (Build.VERSION.SDK_INT >= 26) {
            NotificationManager nm = (NotificationManager) c.getSystemService(Context.NOTIFICATION_SERVICE);
            nm.deleteNotificationChannel(CH_ALARM);
            nm.deleteNotificationChannel(CH_INFO);
        }
        c.getSharedPreferences("r", Context.MODE_PRIVATE).edit()
            .putString("timerTone", timerUri == null ? "" : timerUri)
            .putString("notifTone", notifUri == null ? "" : notifUri).apply();
        channels(c);
    }

    static void show(Context c, int id, String title, String body, boolean alarm) {
        channels(c);
        if (body == null) body = "";
        NotificationManager nm = (NotificationManager) c.getSystemService(Context.NOTIFICATION_SERVICE);
        Notification.Builder b;
        if (Build.VERSION.SDK_INT >= 26) {
            b = new Notification.Builder(c, alarm ? CH_ALARM : CH_INFO);
        } else {
            b = new Notification.Builder(c);
            if (alarm) {
                b.setSound(RingtoneManager.getDefaultUri(RingtoneManager.TYPE_ALARM));
                b.setVibrate(new long[]{0, 500, 250, 500, 250, 500});
                b.setPriority(Notification.PRIORITY_HIGH);
            }
        }
        PendingIntent open = PendingIntent.getActivity(c, 0, new Intent(c, MainActivity.class),
            PendingIntent.FLAG_IMMUTABLE | PendingIntent.FLAG_UPDATE_CURRENT);
        b.setSmallIcon(com.sapp.planner.R.drawable.ic_notification)
         .setContentTitle(title)
         .setContentText(body)
         .setStyle(new Notification.BigTextStyle().bigText(body))
         .setAutoCancel(true)
         .setContentIntent(open);
        if (alarm) b.setCategory(Notification.CATEGORY_ALARM);
        nm.notify(id, b.build());
    }

    static PendingIntent pi(Context c, int id, String kind, String t, String b) {
        Intent i = new Intent(c, ReminderReceiver.class);
        i.putExtra("k", kind);
        i.putExtra("t", t);
        i.putExtra("b", b);
        i.putExtra("i", id);
        return PendingIntent.getBroadcast(c, id, i, PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);
    }

    static void arm(Context c, long at, PendingIntent p) {
        AlarmManager am = (AlarmManager) c.getSystemService(Context.ALARM_SERVICE);
        try {
            if (Build.VERSION.SDK_INT >= 31 && !am.canScheduleExactAlarms()) {
                am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, p);
            } else {
                am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, p);
            }
        } catch (SecurityException e) {
            am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, p);
        }
    }

    static void timer(Context c, long end, String label) {
        arm(c, end, pi(c, TIMER_ID, "timer", label + " completed", "Time is up. Take a break."));
    }

    static void cancelTimer(Context c) {
        AlarmManager am = (AlarmManager) c.getSystemService(Context.ALARM_SERVICE);
        am.cancel(pi(c, TIMER_ID, "timer", "", ""));
    }

    static void schedule(Context c, String json) {
        try {
            AlarmManager am = (AlarmManager) c.getSystemService(Context.ALARM_SERVICE);
            SharedPreferences sp = c.getSharedPreferences("r", Context.MODE_PRIVATE);
            int old = sp.getInt("n", 0);
            for (int i = 0; i < old; i++) am.cancel(pi(c, i, "rem", "", ""));
            JSONArray a = new JSONArray(json);
            int n = Math.min(a.length(), 60);
            long now = System.currentTimeMillis();
            for (int i = 0; i < n; i++) {
                JSONObject o = a.getJSONObject(i);
                long at = o.getLong("at");
                if (at > now) arm(c, at, pi(c, i, "rem", o.getString("t"), o.optString("b", "")));
            }
            sp.edit().putInt("n", n).putString("json", json).apply();
        } catch (Exception ignored) { }
    }

    @Override
    public void onReceive(Context c, Intent i) {
        String k = i.getStringExtra("k");
        if ("timer".equals(k) && MainActivity.fg) return;
        show(c, i.getIntExtra("i", 0), i.getStringExtra("t"), i.getStringExtra("b"), true);
    }

    public static class Boot extends BroadcastReceiver {
        @Override
        public void onReceive(Context c, Intent i) {
            schedule(c, c.getSharedPreferences("r", Context.MODE_PRIVATE).getString("json", "[]"));
        }
    }
}

function evalMath(expr){
  var s=expr.replace(/\s+/g,''),i=0;
  function primary(){
    var m,v;
    if(s[i]==='('){i++;v=expression();if(s[i]===')')i++;return v;}
    m=/^(\d+\.?\d*|\.\d+)/.exec(s.slice(i));
    if(m){i+=m[0].length;return parseFloat(m[0]);}
    m=/^[a-z]+/.exec(s.slice(i));
    if(m){
      var fn=m[0];i+=fn.length;
      if(fn==='pi')return Math.PI;
      if(fn==='e')return Math.E;
      if(s[i]!=='(')throw 0;
      i++;v=expression();if(s[i]===')')i++;
      var rad=calc.deg?(Math.PI/180):1;
      if(fn==='sin')return Math.sin(v*rad);
      if(fn==='cos')return Math.cos(v*rad);
      if(fn==='tan')return Math.tan(v*rad);
      if(fn==='ln')return Math.log(v);
      if(fn==='log')return Math.log10(v);
      if(fn==='sqrt')return Math.sqrt(v);
    }
    throw 0;
  }
  function factor(){
    var v=primary();
    while(s[i]==='^'||s[i]==='!'){
      if(s[i]==='^'){i++;v=Math.pow(v,primary());}
      else if(s[i]==='!'){i++;var f=1;for(var j=2;j<=v;j++)f*=j;v=f;}
    }
    return v;
  }
  function term(){
    var v=factor();
    while(s[i]==='*'||s[i]==='/'||s[i]==='%'){
      var op=s[i++];var r=factor();
      v=op==='*'?v*r:op==='/'?v/r:v*(r/100);
    }
    return v;
  }
  function expression(){
    var v=term();
    while(s[i]==='+'||s[i]==='-'){
      var op=s[i++];var r=term();
      v=op==='+'?v+r:v-r;
    }
    return v;
  }
  var res=expression();
  if(!isFinite(res))throw 0;
  return res;
}

function parseSyllabus(txt){
  var lines=String(txt).split(/\n+/);
  var currentHeading='',items=[];
  lines.forEach(function(l){
    var s=l.trim();if(!s)return;
    var headingMatch=s.match(/^(\d+[\.\)]|[A-Z]{2,}\s*:|[#]+)\s*(.*)/i);
    if(headingMatch && !s.includes('[')){
      if(currentHeading && items.length){
        S.notes.unshift({id:uid(),title:currentHeading.slice(0,70),body:'',subs:items.slice(0,50)});
        items=[];
      }
      currentHeading=s;
    } else {
      var clean=s.replace(/^\[\s*\]\s*/,'').replace(/^-\s*/,'').trim();
      if(clean.length>1){
        items.push({id:uid(),t:clean.slice(0,80),done:false});
      }
    }
  });
  if(currentHeading && items.length){
    S.notes.unshift({id:uid(),title:currentHeading.slice(0,70),body:'',subs:items.slice(0,50)});

function loadTess(){
  if(window.Tesseract)return Promise.resolve(window.Tesseract);
  return new Promise(function(res,rej){
    var sc=document.createElement('script');
    sc.src='https://cdn.jsdelivr.net/npm/tesseract.js@5.1.1/dist/tesseract.min.js';
    sc.onload=function(){window.Tesseract?res(window.Tesseract):rej()};
    sc.onerror=rej;
    document.head.appendChild(sc);
  });
}

function readImg(file){
  return new Promise(function(res,rej){
    var rd=new FileReader();
    rd.onload=function(){var im=new Image();im.onload=function(){res(im)};im.onerror=rej;im.src=rd.result};
    rd.onerror=rej;
    rd.readAsDataURL(file);
  });
}

function drawCanvas(im,max){
  var k=Math.min(1,max/Math.max(im.width,im.height)),c=document.createElement('canvas');
  c.width=Math.max(1,Math.round(im.width*k));c.height=Math.max(1,Math.round(im.height*k));
  c.getContext('2d').drawImage(im,0,0,c.width,c.height);
  return c;
}
